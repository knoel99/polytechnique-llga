9. Mise en œuvre : simulations networkx

# TME 1 -- Erdos-Renyi vs Barabasi-Albert : lois de degres et composante geante
import numpy as np, networkx as nx
rng = np.random.default_rng(0)

G_er  = nx.gnp_random_graph(2000, 6/2000, seed=1)      # c = 6
G_ba  = nx.barabasi_albert_graph(2000, 3, seed=1)      # m = 3, degre moyen 6
for name, G in (("ER", G_er), ("BA", G_ba)):
    ccs = sorted((len(c) for c in nx.connected_components(G)), reverse=True)
    degs = [d for _, d in G.degree()]
    print(f"{name}: plus grande composante {ccs[0]/G.number_of_nodes():.2f}"
          f" | degre max {max(degs)} | clustering {nx.average_clustering(G):.4f}")
# attendu (theoreme 1.3 + 2.5): ER ~ 0.98, degre max ~ 15 ; BA ~ 1.0, degre max ~ 80
# verification du rho de 1.3 : rho solution de rho = 1 - exp(-6 rho) ~ 0.997
# TME 2 -- SIR a temps discret : le seuil depend-il de la loi des degres ?
def sir(G, p, patient0):
    """SIR discret : infection p par arete et par pas, guerison en 1 pas.
    Renvoie la taille finale de l'epidemie."""
    inf, seen = {patient0}, {patient0}
    while inf:
        new = set()
        for u in inf:
            new |= {v for v in G[u] if v not in seen and rng.random() < p}
        seen |= new
        inf = new
    return len(seen)

def outbreak(G, p, trials=200):
    n = G.number_of_nodes()
    sizes = [sir(G, p, int(rng.integers(n))) for _ in range(trials)]
    return np.mean(sizes) / n            # fraction finale de la population

for tau in (0.3, 0.5, 0.8, 1.2, 2.0):   # tau = beta/delta ; guerison en 1 pas,
    p = 1 - np.exp(-tau)                 # infection par arete et par pas
    print(f"tau={tau:4.1f}  ER {outbreak(G_er, p):.3f}   BA {outbreak(G_ba, p):.3f}")
# exploration guidee : tracer la fraction finale en fonction de tau pour ER et BA,
# et verifier le theoreme 3.5 : la courbe BA decolle bien avant la courbe ER
# (seuil nul sur les reseaux scale-free).
# TME 3 -- SBM : detection spectrale (adjacence en dense, non-retour en creux)
from scipy.sparse.linalg import eigsh

sizes, p_in, p_out = [500, 500], 0.10, 0.02
G = nx.stochastic_block_model(sizes, [[p_in, p_out], [p_out, p_in]], seed=3)
A = nx.to_scipy_sparse_array(G).astype(float)
truth = np.concatenate([np.zeros(sizes[0]), np.ones(sizes[1])])

v2 = eigsh(A, k=2, which="LA")[1][:, 1]              # 2e vecteur propre
hat = (v2 > np.median(v2)).astype(int)
acc = max(np.mean(hat == truth), np.mean(hat != truth))
print("accuracy (adjacence, dense):", round(acc, 3))   # attendu ~ 1.0
