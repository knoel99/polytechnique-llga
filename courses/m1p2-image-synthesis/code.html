Un path tracer complet en Python

Le programme ci-dessous (~70 lignes, NumPy seul) met en œuvre les chapitres 4 (caméra), 6 (échantillonnage cosinus, roulette par profondeur), 1–3 (radiance, Lambert, ombres) et 5/9 (rebonds : série du théorème 2.2). Réduire W, H, SPP pour tester ; en pleine résolution, compter ~1 minute (interpréteur) : chaque pixel est un estimateur MC indépendant — la parallélisation est triviale.

import numpy as np

W, H, SPP = 240, 180, 8
def norm(v): return v / np.linalg.norm(v)

# scene : spheres (centre, rayon, albedo, albédo rho)
S = [(( 0.0, -100.0, -1.0), 100.0, (0.85, 0.80, 0.75), 0.6),   # sol
     (( 0.0,    1.0, -5.0),   1.0, (0.75, 0.25, 0.20), 0.7),
     ((-2.3,    0.7, -4.2),   0.7, (0.25, 0.70, 0.30), 0.7),
     (( 2.3,    0.7, -4.2),   0.7, (0.25, 0.35, 0.90), 0.7)]
Lp  = np.array([6.0, 9.0, 1.0])      # lumiere ponctuelle
I   = np.array([180.0, 170.0, 150.0])
SKY = np.array([0.45, 0.60, 0.85])   # radiance du ciel

def intersect(o, d):
    tbest, ibest = 1e9, -1           # premiere intersection (ch. 7 naive)
    for i, (c, r, _, _) in enumerate(S):
        oc = o - c; b = oc @ d; disc = b*b - (oc @ oc - r*r)
        if disc > 0:
            t = -b - np.sqrt(disc)
            if 1e-3 < t < tbest: tbest, ibest = t, i
    return tbest, ibest

def trace(o, d):
    col, att = np.zeros(3), np.ones(3)          # L = somme des T^k Le (th. 2.2)
    for depth in range(4):                      # ... + roulette implicite
        t, i = intersect(o, d)
        if i < 0:
            return col + att * SKY * max(0.1, d[1])
        c, r, albedo, rho = S[i]
        p = o + t*d; n = norm(p - c)
        # next event estimation : lumiere ponctuelle + rayon d'ombre (ch. 6)
        Lv = Lp - p; dist = np.linalg.norm(Lv); L = Lv / dist
        if intersect(p + 1e-3*n, L)[0] > dist:
            col += att * rho * albedo * I / dist**2 * max(0.0, n @ L)
        # rebond diffus : echantillonnage cosinus (p = cos/pi, th. 6.3)
        d = norm(n + np.random.normal(size=3))
        o, att = p + 1e-3*n, att * rho * albedo
    return col

img = np.zeros((H, W, 3))
for y in range(H):
    for x in range(W):
        acc = np.zeros(3)
        for _ in range(SPP):                    # AA stochastique (ch. 8)
            u = (x + np.random.rand()) / W  - 0.5
            v = 0.5 - (y + np.random.rand()) / H
            acc += trace(np.array([0.0, 1.2, 0.0]), norm([u, v, -1.0]))
        img[y, x] = acc / SPP

img = np.clip(img, 0, None) ** (1.0 / 2.2)      # tone map gamma (ch. 8)
img = img / img.max()                           # normalisation d'exposition
with open("render.ppm", "wb") as f:
    f.write(f"P6\n{W} {H}\n255\n".encode())
    f.write((img * 255).astype(np.uint8).tobytes())
Exercices de lecture du code (1) La ligne d = norm(n + gaussienne) est l'échantillonnage cosinus du théorème 6.3 — pourquoi ce simple re-centrage réalise-t-il la densité \(\cos\theta/\pi\) (indice : distribution de la projection d'une gaussienne isotrope sur l'hémisphère) ? (2) L'atténuation att est le produit des facteurs de transport divisés par les probabilités — identifier chaque terme de la définition 6.1. (3) Remplacer la lumière ponctuelle par une sphère émissive et échantillonner sa surface : le bruit des ombres s'effondre (NEE sur lumière étendue, définition 6.4).
