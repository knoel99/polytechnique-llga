A complete path tracer in Python

The program below (~70 lines, NumPy only) implements chapters 4 (camera), 6 (cosine sampling, depth roulette), 1–3 (radiance, Lambert, shadows) and 5/9 (bounces: the series of Theorem 2.2). Reduce W, H, SPP to test; at full resolution, expect ~1 minute (interpreter): each pixel is an independent MC estimator — parallelization is trivial.

import numpy as np

W, H, SPP = 240, 180, 8
def norm(v): return v / np.linalg.norm(v)

# scene: spheres (center, radius, albedo, albedo rho)
S = [(( 0.0, -100.0, -1.0), 100.0, (0.85, 0.80, 0.75), 0.6),   # ground
     (( 0.0,    1.0, -5.0),   1.0, (0.75, 0.25, 0.20), 0.7),
     ((-2.3,    0.7, -4.2),   0.7, (0.25, 0.70, 0.30), 0.7),
     (( 2.3,    0.7, -4.2),   0.7, (0.25, 0.35, 0.90), 0.7)]
Lp  = np.array([6.0, 9.0, 1.0])      # point light
I   = np.array([180.0, 170.0, 150.0])
SKY = np.array([0.45, 0.60, 0.85])   # sky radiance

def intersect(o, d):
    tbest, ibest = 1e9, -1           # first intersection (naive ch. 7)
    for i, (c, r, _, _) in enumerate(S):
        oc = o - c; b = oc @ d; disc = b*b - (oc @ oc - r*r)
        if disc > 0:
            t = -b - np.sqrt(disc)
            if 1e-3 < t < tbest: tbest, ibest = t, i
    return tbest, ibest

def trace(o, d):
    col, att = np.zeros(3), np.ones(3)          # L = sum of T^k Le (th. 2.2)
    for depth in range(4):                      # ... + implicit roulette
        t, i = intersect(o, d)
        if i < 0:
            return col + att * SKY * max(0.1, d[1])
        c, r, albedo, rho = S[i]
        p = o + t*d; n = norm(p - c)
        # next event estimation: point light + shadow ray (ch. 6)
        Lv = Lp - p; dist = np.linalg.norm(Lv); L = Lv / dist
        if intersect(p + 1e-3*n, L)[0] > dist:
            col += att * rho * albedo * I / dist**2 * max(0.0, n @ L)
        # diffuse bounce: cosine sampling (p = cos/pi, th. 6.3)
        d = norm(n + np.random.normal(size=3))
        o, att = p + 1e-3*n, att * rho * albedo
    return col

img = np.zeros((H, W, 3))
for y in range(H):
    for x in range(W):
        acc = np.zeros(3)
        for _ in range(SPP):                    # stochastic AA (ch. 8)
            u = (x + np.random.rand()) / W  - 0.5
            v = 0.5 - (y + np.random.rand()) / H
            acc += trace(np.array([0.0, 1.2, 0.0]), norm([u, v, -1.0]))
        img[y, x] = acc / SPP

img = np.clip(img, 0, None) ** (1.0 / 2.2)      # gamma tone map (ch. 8)
img = img / img.max()                           # exposure normalization
with open("render.ppm", "wb") as f:
    f.write(f"P6\n{W} {H}\n255\n".encode())
    f.write((img * 255).astype(np.uint8).tobytes())
Code reading exercises (1) The line d = norm(n + gaussian) is the cosine sampling of Theorem 6.3 — why does this simple re-centering realize the density \(\cos\theta/\pi\) (hint: distribution of the projection of an isotropic Gaussian onto the hemisphere)? (2) The attenuation att is the product of transport factors divided by probabilities — identify each term of Definition 6.1. (3) Replace the point light with an emissive sphere and sample its surface: the shadow noise collapses (NEE on an extended light, Definition 6.4).
