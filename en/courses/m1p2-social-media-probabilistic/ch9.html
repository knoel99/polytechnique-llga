9. Hands-on: networkx simulations

# TME 1 -- Erdos-Renyi vs Barabasi-Albert: degree laws and giant component
import numpy as np, networkx as nx
rng = np.random.default_rng(0)

G_er  = nx.gnp_random_graph(2000, 6/2000, seed=1)      # c = 6
G_ba  = nx.barabasi_albert_graph(2000, 3, seed=1)      # m = 3, mean degree 6
for name, G in (("ER", G_er), ("BA", G_ba)):
    ccs = sorted((len(c) for c in nx.connected_components(G)), reverse=True)
    degs = [d for _, d in G.degree()]
    print(f"{name}: largest component {ccs[0]/G.number_of_nodes():.2f}"
          f" | max degree {max(degs)} | clustering {nx.average_clustering(G):.4f}")
# expected (theorems 1.3 + 2.5): ER ~ 0.98, max degree ~ 15; BA ~ 1.0, max degree ~ 80
# check the rho of 1.3: rho solution of rho = 1 - exp(-6 rho) ~ 0.997
# TME 2 -- discrete-time SIR: does the threshold depend on the degree law?
def sir(G, p, patient0):
    """Discrete SIR: infection p per edge per step, recovery in 1 step.
    Returns the final size of the epidemic."""
    inf, seen = {patient0}, {patient0}
    while inf:
        new = set()
        for u in inf:
            new |= {v for v in G[u] if v not in seen and rng.random() < p}
        seen |= new
        inf = new
    return len(seen)

def outbreak(G, p, trials=200):
    n = G.number_of_nodes()
    sizes = [sir(G, p, int(rng.integers(n))) for _ in range(trials)]
    return np.mean(sizes) / n            # final fraction of the population

for tau in (0.3, 0.5, 0.8, 1.2, 2.0):   # tau = beta/delta; recovery in 1 step,
    p = 1 - np.exp(-tau)                 # infection per edge per step
    print(f"tau={tau:4.1f}  ER {outbreak(G_er, p):.3f}   BA {outbreak(G_ba, p):.3f}")
# guided exploration: plot the final fraction as a function of tau for ER and BA,
# and check theorem 3.5: the BA curve takes off well before the ER curve
# (vanishing threshold on scale-free networks).
# TME 3 -- SBM: spectral detection (adjacency in dense, non-backtracking in sparse)
from scipy.sparse.linalg import eigsh

sizes, p_in, p_out = [500, 500], 0.10, 0.02
G = nx.stochastic_block_model(sizes, [[p_in, p_out], [p_out, p_in]], seed=3)
A = nx.to_scipy_sparse_array(G).astype(float)
truth = np.concatenate([np.zeros(sizes[0]), np.ones(sizes[1])])

v2 = eigsh(A, k=2, which="LA")[1][:, 1]              # 2nd eigenvector
hat = (v2 > np.median(v2)).astype(int)
acc = max(np.mean(hat == truth), np.mean(hat != truth))
print("accuracy (adjacency, dense):", round(acc, 3))   # expected ~ 1.0

TME protocol (guidelines)
  1. Reproducibility: fix the seeds, report mean ± standard deviation over 200 repetitions (Refresher Statistics module, ch. 4-6); plot curves with bootstrap confidence bands.
  2. Check the theories: each TME has an expected value (giant size, takeoff threshold, accuracy): a systematic gap = bug or misreading of the regime — the report must decide.
